import React, { useState, useEffect, useMemo } from 'react';
import { 
  CreditCard, Calendar, Upload, PlusCircle, Trash2, 
  PieChart as PieChartIcon, BarChart3, TrendingUp, AlertTriangle, 
  List, Bot, CheckCircle2, Sparkles, Save, Download, Search, Clock, Wallet, ShieldCheck, ChevronRight,
  Cloud, Edit3, X, Filter, Send, Cpu, Zap, AlertCircle, ArrowUpRight, Check
} from 'lucide-react';
import { db, auth } from './firebase';
import { signInAnonymously, onAuthStateChanged } from 'firebase/auth';
import { collection, doc, setDoc, deleteDoc, onSnapshot, writeBatch } from 'firebase/firestore';
import { PieChart, Pie, Cell, BarChart, Bar, XAxis, YAxis, Tooltip, ResponsiveContainer, Legend } from 'recharts';
import * as XLSX from 'xlsx';

const CATEGORY_OPTIONS = [
  '1) 餐飲', '2) 交通', '3) 八達通增值', '4) 購物', 
  '5) 網購', '6) 管理費', '7) 稅', '8) 貸款', '9) 其他'
];

const CATEGORY_COLORS = {
  '1) 餐飲': '#fbbf24', '2) 交通': '#38bdf8', '3) 八達通增值': '#2dd4bf',
  '4) 購物': '#f472b6', '5) 網購': '#c084fc', '6) 管理費': '#34d399',
  '7) 稅': '#f87171', '8) 貸款': '#a78bfa', '9) 其他': '#94a3b8'
};

const BANK_OPTIONS = ['花旗銀行', '渣打銀行', '恆生銀行', '滙豐銀行', '中國銀行', '其他銀行'];

export default function App() {
  const [activeTab, setActiveTab] = useState('overview');
  const [toast, setToast] = useState({ show: false, msg: '', type: 'info' });

  // 雲端實時狀態
  const [cards, setCards] = useState([]);
  const [transactions, setTransactions] = useState([]);
  const [userId, setUserId] = useState(null);

  // 表單狀態
  const [newBank, setNewBank] = useState('花旗銀行');
  const [newCardName, setNewCardName] = useState('');
  const [newDueDate, setNewDueDate] = useState(() => new Date().toISOString().split('T')[0]);
  const [newRepaymentAmount, setNewRepaymentAmount] = useState('');

  // AI 狀態
  const [chatInput, setChatInput] = useState('');
  const [chatMessages, setChatMessages] = useState([
    { role: 'assistant', text: '你好！我是 FinPulse AI 智能理財顧問。我已成功接通您的雲端帳單，您可以問我「財務狀況」、「還款策略」或「哪張卡欠最多？」' }
  ]);

  const showToast = (msg, type = 'info') => {
    setToast({ show: true, msg, type });
    setTimeout(() => setToast({ show: false, msg: '', type: 'info' }), 3000);
  };

  // 🔐 實時雲端監聽核心 (Real-time Sync)
  useEffect(() => {
    signInAnonymously(auth)
      .then((cred) => setUserId(cred.user.uid))
      .catch(() => showToast('❌ 認證失敗，請檢查 Firebase 規則。', 'error'));

    const unsubscribe = onAuthStateChanged(auth, (user) => {
      if (user) {
        // 實時同步信用卡
        const unsubscribeCards = onSnapshot(collection(db, 'users', user.uid, 'cards'), (snapshot) => {
          setCards(snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() })));
        });

        // 實時同步簽賬明細
        const unsubscribeTx = onSnapshot(collection(db, 'users', user.uid, 'transactions'), (snapshot) => {
          const txData = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
          txData.sort((a, b) => b.date.localeCompare(a.date));
          setTransactions(txData);
        });

        return () => { unsubscribeCards(); unsubscribeTx(); };
      }
    });

    return () => unsubscribe();
  }, []);

  // 新增卡片至 Firestore
  const handleAddCard = async (e) => {
    e.preventDefault();
    if (!newCardName.trim() || !newRepaymentAmount) return showToast('請填寫完整卡片資料', 'warning');
    if (!userId) return showToast('尚未接通雲端', 'error');

    const cardId = `card-${Date.now()}`;
    try {
      await setDoc(doc(db, 'users', userId, 'cards', cardId), {
        bank: newBank,
        name: newCardName.trim(),
        dueDate: newDueDate,
        repaymentAmount: parseFloat(newRepaymentAmount) || 0,
        isPaid: false
      });
      setNewCardName(''); setNewRepaymentAmount('');
      showToast('✅ 卡片已成功同步至 Firebase 雲端資料庫！', 'success');
    } catch (err) { showToast('雲端儲存失敗', 'error'); }
  };

  // 刪除卡片
  const handleDeleteCard = async (id) => {
    if (!userId) return;
    try {
      await deleteDoc(doc(db, 'users', userId, 'cards', id));
      showToast('🗑️ 已從雲端移除該卡片紀錄', 'info');
    } catch (err) { showToast('刪除失敗', 'error'); }
  };

  // 本地 AI 智能診斷核心
  const handleSendMessage = () => {
    const query = chatInput.trim();
    if (!query) return;

    const newMsgs = [...chatMessages, { role: 'user', text: query }];
    setChatMessages(newMsgs);
    setChatInput('');

    setTimeout(() => {
      let reply = "收到您的提問。建議優先繳清高金額或即將到期的帳單，避免產生循環利息。";
      const totalAmount = cards.reduce((sum, c) => sum + (c.repaymentAmount || 0), 0);
      
      if (query.includes("狀況") || query.includes("財務") || query.includes("多少")) {
        reply = `📊 【雲端實時診斷】您本期在雲端資料庫共有 ${cards.length} 張卡片帳單，累計應還總金額為 HK$${totalAmount.toLocaleString()}。`;
      } else if (query.includes("還款") || query.includes("建議") || query.includes("策略")) {
        reply = `💡 【智慧理財還款法】建議採用雪崩還款法：鎖定本期金額最高的卡片集中火力全額全還款，其餘卡片保持最低還款，能最化減少利息負擔。`;
      }
      setChatMessages([...newMsgs, { role: 'assistant', text: reply }]);
    }, 400);
  };

  // 數據即時統計快取 (Memoized Stats)
  const stats = useMemo(() => {
    const totalRepayment = cards.reduce((sum, c) => sum + (Number(c.repaymentAmount) || 0), 0);
    const unpaidCardsCount = cards.filter(c => !c.isPaid).length;
    return { totalRepayment, unpaidCardsCount };
  }, [cards]);

  return (
    <div className="min-h-screen bg-[#0f172a] text-slate-100 p-4 md:p-8">
      {toast.show && (
        <div className="fixed top-5 right-5 z-50 flex items-center gap-2 bg-slate-800 border border-cyan-500/40 text-cyan-200 px-5 py-3 rounded-xl shadow-2xl">
          <Sparkles className="w-4 h-4 text-cyan-400" />
          <span className="text-sm font-semibold">{toast.msg}</span>
        </div>
      )}

      <div className="max-w-6xl mx-auto space-y-6">
        {/* 頂欄系統標頭 */}
        <header className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 bg-slate-900/60 p-6 rounded-2xl border border-slate-800">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-cyan-500 to-indigo-600 flex items-center justify-center font-black">FP</div>
            <div>
              <h1 className="text-xl font-black tracking-wider text-transparent bg-clip-text bg-gradient-to-r from-cyan-400 to-indigo-300">FIN-PULSE v4</h1>
              <p className="text-xs text-slate-400">信用卡管理 Dashboard (Vite 本地獨立編譯版)</p>
            </div>
          </div>
          <div className="flex gap-2">
            <button onClick={() => setActiveTab('overview')} className={`px-4 py-2 rounded-xl text-xs font-bold ${activeTab === 'overview' ? 'bg-cyan-500 text-slate-950' : 'bg-slate-800 text-slate-400'}`}>數據總覽</button>
            <button onClick={() => setActiveTab('ai')} className={`px-4 py-2 rounded-xl text-xs font-bold ${activeTab === 'ai' ? 'bg-purple-500 text-white' : 'bg-slate-800 text-slate-400'}`}>AI 顧問</button>
          </div>
        </header>

        {activeTab === 'overview' ? (
          <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
            {/* 左側與中央：手動錄入與卡片清單 */}
            <div className="lg:col-span-2 space-y-6">
              {/* 指標方塊 */}
              <div className="grid grid-cols-2 gap-4">
                <div className="bg-slate-900/40 p-5 rounded-2xl border border-slate-800">
                  <p className="text-xs text-slate-400 font-bold">本期待繳總額</p>
                  <h3 className="text-2xl font-black mt-1 text-cyan-300">HK${stats.totalRepayment.toLocaleString()}</h3>
                </div>
                <div className="bg-slate-900/40 p-5 rounded-2xl border border-slate-800">
                  <p className="text-xs text-slate-400 font-bold">待繳卡片數量</p>
                  <h3 className="text-2xl font-black mt-1 text-rose-400">{stats.unpaidCardsCount} 張</h3>
                </div>
              </div>

              {/* 手動輸入表單 */}
              <form onSubmit={handleAddCard} className="bg-slate-900/40 p-6 rounded-2xl border border-slate-800 space-y-4">
                <h3 className="text-sm font-bold text-cyan-400 flex items-center gap-1.5"><PlusCircle className="w-4 h-4" /> 手動錄入全新月結單項目</h3>
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <div>
                    <label className="text-[11px] text-slate-400 block mb-1">發卡銀行</label>
                    <select value={newBank} onChange={e => setNewBank(e.target.value)} className="w-full p-2.5 bg-slate-950 border border-slate-800 rounded-xl text-xs text-white">
                      {BANK_OPTIONS.map(b => <option key={b} value={b}>{b}</option>)}
                    </select>
                  </div>
                  <div>
                    <label className="text-[11px] text-slate-400 block mb-1">卡片/帳單名稱</label>
                    <input type="text" placeholder="例如: Cash Back" value={newCardName} onChange={e => setNewCardName(e.target.value)} className="w-full p-2.5 bg-slate-950 border border-slate-800 rounded-xl text-xs text-white" />
                  </div>
                  <div>
                    <label className="text-[11px] text-slate-400 block mb-1">還款到期日</label>
                    <input type="date" value={newDueDate} onChange={e => setNewDueDate(e.target.value)} className="w-full p-2.5 bg-slate-950 border border-slate-800 rounded-xl text-xs text-white" />
                  </div>
                  <div>
                    <label className="text-[11px] text-slate-400 block mb-1">應還金額 (HK$)</label>
                    <input type="number" placeholder="2500" value={newRepaymentAmount} onChange={e => setNewRepaymentAmount(e.target.value)} className="w-full p-2.5 bg-slate-950 border border-slate-800 rounded-xl text-xs text-white" />
                  </div>
                </div>
                <button type="submit" className="w-full bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-bold py-2.5 rounded-xl text-xs transition-colors">確認同步寫入雲端資料庫</button>
              </form>

              {/* 實時信用卡清單 */}
              <div className="space-y-3">
